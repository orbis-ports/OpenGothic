# The tessellation run. Copy to /data/orbis-env.txt (the old /data/tempest-env.txt is still read,
# and is deprecated as of 2026-09-18 - see tempest-env.example.txt for why the shared name won).
#
# ⚠ THIS FILE TURNS TESSELLATION ON, WHICH tempest-env.example.txt CALLS "NOT AN OPTION" TO LEAVE OFF.
# That warning was written when the stage faulted. It faulted because the tessellation FACTORS were
# written through a ring descriptor into a buffer of ours and READ through VGT_TF_MEMORY_BASE, which on
# this console still held Sony's 0xff0000000 - an address nothing had mapped. Measured 2026-08-31:
# dEQP-VK.tessellation.tesscoord.triangles_equal_spacing, the first tessellated draw this port has ever
# issued, died on a texture-cache READ at 0xff0010000.
#
# ORBIS_TF_SONY_BASE backs that address with real memory and moves the ring's DESCRIPTOR and its
# REGISTER both there, so the write path and the read path agree whether or not our register write
# survives. With it, the same 44 cases that died went 44/44 Pass, and the 535-case group came back with
# every failure carrying a geometry shader and none without one. OpenGothic ships no geometry shader.
#
# ⚠ WHAT IS NOT ESTABLISHED, and it is exactly what this run is for. Every dEQP measurement above was
# taken on tiny patches: over 3320 submissions the offchip ring was touched for ZERO bytes, because
# RADV kept the TCS outputs in LDS, and the factor ring was never written past its programmed size.
# This title is not that. main.tesc computes its factors from screen-space edge length - `l*16+1`, on
# every visible triangle - and passes a real Varyings payload per vertex. Both rings will be used the
# way the hardware actually uses them for the first time.
#
# So the watermark is on: it reports every 256th submission what the hardware reached, and says so
# even when the answer is zero.
#
# SPDX-License-Identifier: MIT

# ⚠ A NAME OF ITS OWN. The example file's warning, paid for once already: a run of 2026-08-23 wrote
# over 4.3 MB of CTS baseline because the copy on the console carried a deqp log name.
MESA_LOG_FILE=/data/og-tess.log
MESA_LOG_LEVEL=info

# The driver defaults tessellation OFF and advertises nothing; this is the line that turns it on.
ORBIS_NO_TESS=0

# The repair. Without it the first tessellated draw faults at 0xff0000000 + a small offset.
ORBIS_TF_SONY_BASE=1

# The instrument. Poisons everything past VGT_TF_RING_SIZE inside Sony's 2 MiB mapping and reports the
# highest byte the hardware disturbed, every 256th submission. Nothing else in the process can reach
# that address, so a disturbed byte is a measurement rather than a reading.
ORBIS_TESS_RING_WATERMARK=256

# frames with it out of the environment and no faults, and the console's env file is currently renamed
# .off - so the title's normal state is no environment file at all. Adding the workaround back here
# would make this run differ from the shipping one in two ways instead of one.
